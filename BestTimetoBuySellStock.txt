import java.util.*;
import java.util.Iterator;
import java.util.stream.*;

public class BestTimeBuySellStock {

	public static Integer BestTimebuysell(Integer[] arr) {

		Integer BuyStock = arr[0];
		Integer max = 0;

		for (int i = 1; i < arr.length; i++) {

			if (BuyStock >= arr[i]) {

				BuyStock = arr[i];

			}

			max = Math.max(max, arr[i] - BuyStock);

		}

		return max;

	}

	public static void main(String args[]) {

		BestTimeBuySellStock obj = new BestTimeBuySellStock();

		//Integer[] arr = { 7, 1, 5, 3, 6, 4 };
		Integer[] arr = {7,6,4,3,1};

		Integer result = obj.BestTimebuysell(arr);

		System.out.println(result);

	}

}
